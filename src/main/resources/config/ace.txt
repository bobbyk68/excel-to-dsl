private static String lastGroup(AtomicHit hit) {
    if (hit == null || hit.groups() == null || hit.groups().isEmpty()) return null;
    String g = hit.groups().get(hit.groups().size() - 1);
    return g == null ? null : g.replace("\"","").trim();
}
private static java.util.List<String> groupsToList(AtomicHit hit) {
    if (hit == null || hit.groups() == null || hit.groups().isEmpty()) return java.util.Collections.emptyList();
    if (hit.groups().size() > 1) {
        java.util.List<String> xs = new java.util.ArrayList<>();
        for (String g : hit.groups()) if (g != null && !g.isBlank()) xs.add(g.replace("\"","").trim());
        return xs;
    }
    String csv = hit.groups().get(0);
    if (csv == null || csv.isBlank()) return java.util.Collections.emptyList();
    java.util.List<String> xs = new java.util.ArrayList<>();
    for (String p : csv.split(",")) {
        String v = p.replace("\"","").trim();
        if (!v.isBlank()) xs.add(v);
    }
    return xs;
}
private static java.util.List<String> groupsToListOrFallback(AtomicHit hit, RuleRow row) {
    var xs = groupsToList(hit);
    if (!xs.isEmpty()) return xs;
    var merged = row == null ? null : row.mergedThencodes();
    if (merged == null) return java.util.Collections.emptyList();
    java.util.List<String> out = new java.util.ArrayList<>();
    for (String s : merged) if (s != null && !s.isBlank()) out.add(s.replace("\"","").trim());
    return out;
}
