// Minimal adapter so emitXxx(...) doesn't depend directly on DslWriter.
final class LocalDslBuffer {
    private final java.util.List<String> whens = new java.util.ArrayList<>();
    private final java.util.List<String> thens = new java.util.ArrayList<>();

    void when(String line) { if (line != null && !line.isBlank()) whens.add(line); }
    void then(String line) { if (line != null && !line.isBlank()) thens.add(line); }

    boolean isEmpty() { return whens.isEmpty() && thens.isEmpty(); }

    // Push to your real DSL writer using your existing API
    void flushTo(DslWriter dsl) {
        if (isEmpty()) return;
        dsl.when(whens);          // if you have a bulk method
        dsl.then(thens);          // or loop and call dsl.when(..) individually
    }

    // Handy if you want a single string for logging/tests
    String asBlock() {
        StringBuilder sb = new StringBuilder("when\n");
        for (String w : whens) sb.append("  ").append(w).append('\n');
        sb.append("then\n");
        for (String t : thens) sb.append("  ").append(t).append('\n');
        return sb.toString();
    }
}
