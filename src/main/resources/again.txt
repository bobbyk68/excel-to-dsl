package uk.gov.hmrc.pattern;

import java.util.*;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

/**
 * Bridges HyphenTwoPhaseComposer.Result → parsed clauses you can reason about.
 * - Uses PatternIntrospector.parse(...) to classify each English line.
 * - Extracts values like UE, 72M, D005,D008... robustly.
 */
public final class ResultAdapters {

    /** One resolved (parsed) English line. */
    public static final class ResolvedClause {
        private final String raw;                           // original English line
        private final PatternIntrospector.Meta meta;        // op, q, neg, fieldKey, anchor, kind
        private final List<String> values;                  // extracted values (order-preserving, deduped)

        public ResolvedClause(String raw, PatternIntrospector.Meta meta, List<String> values) {
            this.raw = raw; this.meta = meta; this.values = values;
        }
        public String raw() { return raw; }
        public PatternIntrospector.Meta meta() { return meta; }
        public List<String> values() { return values; }
        public String value() { return (values == null || values.isEmpty()) ? null : values.get(0); }

        @Override public String toString() {
            return "ResolvedClause{raw='" + raw + "', op=" + meta.operator() + ", q=" + meta.quantifier() +
                   ", neg=" + meta.negated() + ", kind=" + meta.patternKind() +
                   ", field=" + meta.fieldKey() + ", anchor=" + meta.anchorPath() +
                   ", values=" + values + "}";
        }
    }

    /** Turn the composer result into a list of ResolvedClause (one per DSLR line). */
    public static List<ResolvedClause> resolve(HyphenTwoPhaseComposer.Result res) {
        if (res == null || res.dslrLines == null || res.dslrLines.isEmpty()) return Collections.emptyList();
        List<ResolvedClause> out = new ArrayList<ResolvedClause>(res.dslrLines.size());
        for (String line : res.dslrLines) {
            String ln = (line == null) ? "" : line.trim();
            // Defensive: ignore purely decorative bullets like "- "
            if (ln.equals("-") || ln.equals("- ")) continue;

            PatternIntrospector.Meta meta = PatternIntrospector.parse(ln);
            List<String> vals = extractValues(ln);
            out.add(new ResolvedClause(ln, meta, vals));
        }
        return out;
    }

    // ─────────────────────────────────────────────────────────────────────
    // Value extraction helpers (robust, order-preserving, deduping)
    // ─────────────────────────────────────────────────────────────────────

    /** Extract values like "UE", "72M", "D005","D008" from a single English line. */
    static List<String> extractValues(String line) {
        if (line == null) return Collections.emptyList();
        String src = line.trim();
        if (src.isEmpty()) return Collections.emptyList();

        // 1) Prefer quoted tokens: "D005","D008"
        Matcher qm = Pattern.compile("\"([^\"]+)\"").matcher(src);
        List<String> quoted = new ArrayList<String>();
        while (qm.find()) quoted.add(qm.group(1).trim());
        if (!quoted.isEmpty()) return dedupeStable(quoted);

        String lower = src.toLowerCase();
        int idx;

        // 2) After 'equals '
        if ((idx = lower.indexOf("equals ")) >= 0) {
            String tail = src.substring(idx + "equals ".length()).trim();
            if (tail.isEmpty()) return Collections.emptyList(); // avoid ""
            return dedupeStable(splitCodes(tail));
        }

        // 3) After 'one of '
        if ((idx = lower.indexOf("one of ")) >= 0) {
            String tail = src.substring(idx + "one of ".length());
            return dedupeStable(splitCodes(tail));
        }

        // 4) Last ' in ' (word-bounded, catches: " in D005, D008")
        if ((idx = lastIndexOfWord(lower, " in ")) >= 0) {
            String tail = src.substring(idx + " in ".length());
            return dedupeStable(splitCodes(tail));
        }

        return Collections.emptyList();
    }

    /** Return last index of needle with surrounding spaces (simple word boundary). */
    private static int lastIndexOfWord(String haystack, String needleWithSpaces) {
        return haystack.lastIndexOf(needleWithSpaces);
    }

    /** Split tokens like D005, D008, 72M. Allows A–Z, a–z, 0–9, '.', '_', '/', '-' */
    private static List<String> splitCodes(String s) {
        List<String> out = new ArrayList<String>();
        String[] parts = s.split("[,\\s]+");
        for (int i = 0; i < parts.length; i++) {
            String t = parts[i].trim();
            if (!t.isEmpty() && t.matches("[A-Za-z0-9._/-]+")) out.add(t);
        }
        return out;
    }

    /** Keep first occurrences, preserve order. */
    private static List<String> dedupeStable(List<String> in) {
        if (in == null || in.isEmpty()) return Collections.emptyList();
        LinkedHashSet<String> set = new LinkedHashSet<String>(in);
        return new ArrayList<String>(set);
    }
}


List<ResultAdapters.ResolvedClause> ifClauses   = ResultAdapters.resolve(ifRes);
List<ResultAdapters.ResolvedClause> thenClauses = ResultAdapters.resolve(thenRes);
