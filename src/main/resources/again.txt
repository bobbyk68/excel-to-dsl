public interface Emitter {
    // Higher = takes precedence if multiple canHandle(...) return true
    int priority();

    boolean canHandle(EmitContext ctx);

    void emit(EmitContext ctx, DslFileWriter dsl);
}


public final class EmitContext {
    public RuleRow row;            // full Excel-derived row
    public AtomicHit ifHit;        // parsed IF
    public AtomicHit thenHit;      // parsed THEN
    public ConstraintCase kase;    // ALL_OF, ANY_OF, etc.
    public String combinedTemplate;
    public List<String> combinedGroup;
}


public final class EmitterRegistry {
    private final List<Emitter> emitters = new ArrayList<>();

    public EmitterRegistry register(Emitter e) {
        emitters.add(e);
        emitters.sort(Comparator.comparingInt(Emitter::priority).reversed());
        return this;
    }

    public Emitter resolve(EmitContext ctx) {
        for (Emitter e : emitters) if (e.canHandle(ctx)) return e;
        return GenericEmitter.INSTANCE;
    }
}
public final class GenericEmitter implements Emitter {
    public static final GenericEmitter INSTANCE = new GenericEmitter();

    @Override public int priority() { return 0; }

    @Override public boolean canHandle(EmitContext ctx) {
        return ctx.combinedTemplate != null && ctx.combinedGroup != null;
    }

    @Override public void emit(EmitContext ctx, DslFileWriter dsl) {
        // Treat combinedTemplate like: "at least one {anchor}.{field} equals {1}"
        dsl.then(ctx.combinedTemplate, ctx.combinedGroup.toArray(String[]::new));
    }
}



public final class AdditionalDocsEqualsEmitter implements Emitter {

    private static final Pattern AD_CODE_EQ = Pattern.compile(
        "additionalDocuments\\.type\\.code\\s+(?:must\\s+)?equals\\s+([A-Za-z0-9]+)",
        Pattern.CASE_INSENSITIVE);

    @Override public int priority() { return 100; } // outrank generic

    @Override public boolean canHandle(EmitContext ctx) {
        String thenText = ctx.row.thenCondition();
        return thenText != null && thenText.toLowerCase().contains("additionaldocuments");
    }

    @Override public void emit(EmitContext ctx, DslFileWriter dsl) {
        String code = extractCode(ctx);
        if (code == null || code.isBlank()) {
            System.err.println("[AD-EMIT] No code for row " + ctx.row.id());
            return;
        }
        // Minimal facts: anchor + requirement
        dsl.when("there is at least one GoodsItem");
        dsl.then("at least one GoodsItem.additionalDocuments.type.code equals {code}", code);
    }

    private String extractCode(EmitContext ctx) {
        // 1) merged list if truly merged
        List<String> merged = ctx.row.getMergedThenCodes();
        if (merged != null && !merged.isEmpty()) return merged.get(0);

        // 2) regex over THEN English
        Matcher m = AD_CODE_EQ.matcher(ctx.row.thenCondition());
        if (m.find()) return m.group(1);

        // 3) parsed groups
        if (ctx.thenHit != null && ctx.thenHit.groups() != null && !ctx.thenHit.groups().isEmpty())
            return ctx.thenHit.groups().get(ctx.thenHit.groups().size() - 1);

        return null;
    }
}


public final class CountDiffersEmitter implements Emitter {

    @Override public int priority() { return 90; }

    @Override public boolean canHandle(EmitContext ctx) {
        String pre = ctx.row.getCombinedThenCondition();
        return pre != null && pre.contains("count differs");
    }

    @Override public void emit(EmitContext ctx, DslFileWriter dsl) {
        // Example: "Additional information codes {1} count differs from {2}"
        dsl.then(ctx.row.getCombinedThenCondition(), ctx.row.getMergedThenCodes().toArray(String[]::new));
    }
}


public final class RequestedPrevProcedureCompositeEmitter implements Emitter {

    // Specific, but less specific than AD → sits just below AD emitter
    @Override public int priority() { return 95; }

    @Override public boolean canHandle(EmitContext ctx) {
        // We require BOTH:
        //  (A) IF mentions requestedProcedure equals {code}
        //  (B) THEN says previousProcedure code "must is one of" {list}
        String ifText   = safeLower(ofIf(ctx));
        String thenText = safeLower(ofThen(ctx));
        return mentionsRequestedEq(ifText) && mentionsPrevMustOneOf(thenText);
    }

    @Override public void emit(EmitContext ctx, DslFileWriter dsl) {
        // 1) Extract requested code
        String reqCode = extractRequestedCode(ctx);
        // 2) Extract allowed list on previousProcedure
        java.util.List<String> allowed = extractPrevAllowedList(ctx);

        if (isBlank(reqCode) || allowed.isEmpty()) {
            System.err.println("[PROC-EMIT] Missing data for row " + ctx.row.id()
                + " IF='" + ofIf(ctx) + "' THEN='" + ofThen(ctx) + "'");
            return;
        }

        // 3) Emit WHEN block lines (your DSL style)
        dsl.when("Goods item exists");
        dsl.when("- with requested procedure code equals \"{code}\"", reqCode);
        dsl.when("- with previous procedure code not in {csv}", quoteCsv(allowed));
        // No THEN content here; your framework emits the error line elsewhere
    }

    // ─── helpers ────────────────────────────────────────────────────────────────

    // Robust regexes
    private static final java.util.regex.Pattern REQ_EQ =
        java.util.regex.Pattern.compile(
            "requested\\s*procedure(?:\\.code)?\\s*equals\\s*\"?([A-Za-z0-9]+)\"?",
            java.util.regex.Pattern.CASE_INSENSITIVE);

    private static final java.util.regex.Pattern PREV_ONE_OF =
        java.util.regex.Pattern.compile(
            "previous\\s*procedure(?:\\.code)?\\s*must\\s+is\\s+one\\s+of\\s+([A-Za-z0-9,\\s\"]+)",
            java.util.regex.Pattern.CASE_INSENSITIVE);

    private String extractRequestedCode(EmitContext ctx) {
        // Prefer parsed IF hit if present
        if (ctx.ifHit != null && ctx.ifHit.groups() != null && !ctx.ifHit.groups().isEmpty()) {
            String last = ctx.ifHit.groups().get(ctx.ifHit.groups().size() - 1);
            if (!isBlank(last)) return last.replace("\"", "");
        }
        // Fallback: regex over IF text
        java.util.regex.Matcher m = REQ_EQ.matcher(ofIf(ctx));
        return m.find() ? m.group(1) : null;
    }

    private java.util.List<String> extractPrevAllowedList(EmitContext ctx) {
        // Prefer parsed THEN hit groups if they carry the CSV
        if (ctx.thenHit != null && ctx.thenHit.groups() != null && !ctx.thenHit.groups().isEmpty()) {
            String csv = ctx.thenHit.groups().get(ctx.thenHit.groups().size() - 1);
            return normalizeList(csv);
        }
        // Fallback: regex over THEN text
        java.util.regex.Matcher m = PREV_ONE_OF.matcher(ofThen(ctx));
        if (m.find()) return normalizeList(m.group(1));
        return java.util.Collections.emptyList();
    }

    private java.util.List<String> normalizeList(String csv) {
        if (isBlank(csv)) return java.util.Collections.emptyList();
        java.util.List<String> out = new java.util.ArrayList<>();
        for (String s : csv.replace("\"","").split("[,\\s]+")) {
            if (!isBlank(s)) out.add(s);
        }
        return out;
    }

    private String quoteCsv(java.util.List<String> items) {
        return "\"" + String.join("\",\"", items) + "\"";
    }

    private String ofIf(EmitContext ctx)   { return ctx.row != null ? ctx.row.ifCondition()   : null; }
    private String ofThen(EmitContext ctx) { return ctx.row != null ? ctx.row.thenCondition() : null; }

    private String safeLower(String s) { return s == null ? "" : s.toLowerCase(); }
    private boolean isBlank(String s) { return s == null || s.trim().isEmpty(); }

    private boolean mentionsRequestedEq(String s) {
        return s.contains("requestedprocedure") || s.contains("requested procedure");
    }

    private boolean mentionsPrevMustOneOf(String s) {
        return s.contains("previousprocedure") || s.contains("previous procedure");
    }
}
