for (CompiledAtomic ca : compiled) {
    Matcher m = ca.regex().matcher(literal);
    if (m.matches()) {
        List<String> substituted = substituteAll(List.of(ca.dsl()), m);
        List<String> groups = new ArrayList<>(m.groupCount());
        for (int i = 1; i <= m.groupCount(); i++) groups.add(m.group(i).trim());

        // ✳️ NEW: pull semantics from the compiled pattern’s metadata
        FieldKey fieldKey       = ca.fieldKey();        // e.g. AD_TYPE_CODE
        AnchorScope anchorScope = AnchorScope.GI;       // all GoodsItem.* → GI
        String anchorPath       = ca.anchorPath();      // "GoodsItem.additionalDocuments"
        Operator operator       = ca.operator();        // EQ / IN
        Quantifier quantifier   = ca.quantifier();      // EXISTS / ALL / ...
        boolean negated         = ca.negated();         // from “must not…”
        PatternKind patternKind = ca.patternKind();     // MUST_NOT_ONE_OF, etc.

        AtomicHit hit = new AtomicHit(
            ca.dsl(), substituted.get(0), groups, isIf, literal,
            fieldKey, anchorScope, anchorPath, operator, quantifier, negated, patternKind
        );
        cache.put(literal, hit);
        return hit;
    }
