public void collectAll(List<RuleRow> rows, DslWriter dsl) {
    for (RuleRow row : rows) {
        // Compose first (your existing calls)
        AtomicHit ifHit   = matcher.matchIf(row.ifCondition());
        AtomicHit thenHit = matcher.matchThen(row.thenCondition());
        ConstraintCase kase = ConstraintCase.fromEnglish(row.thenCondition());

        HyphenTwoPhaseComposer.Result ifRes   = null;
        HyphenTwoPhaseComposer.Result thenRes = null;

        if (ifHit  != null) ifRes  = composer.composeEnglish(HyphenTwoPhaseComposer.Role.IF,   kase, ifHit.template,  ifHit.groups,  null);
        if (thenHit!= null) thenRes= composer.composeEnglish(HyphenTwoPhaseComposer.Role.THEN, kase, thenHit.template,thenHit.groups,null);

        // Decide using the composed lines (not raw Excel)
        if (touchesAdditionalDocs(ifRes, thenRes)) {
            emitAdditionalDocsFromComposed(ifRes, thenRes, row, dsl);
        } else {
            // Default: emit composed lines as-is
            if (ifRes  != null && ifRes.dslrLines  != null) for (String s : ifRes.dslrLines)  dsl.whenAnchor(s);
            if (thenRes!= null && thenRes.dslrLines!= null && !thenRes.dslrLines.isEmpty())
                dsl.thenAnchor(thenRes.dslrLines.get(0)); // or loop if you normally emit more
        }

        dsl.endRule(row.id());
    }
}


// Coalesce additional-docs across IF+THEN *after* compose.
// Emits: WHEN header + IF dashes + THEN (negated) dashes + THEN header.
private void emitAdditionalDocsFromComposed(HyphenTwoPhaseComposer.Result ifRes,
                                            HyphenTwoPhaseComposer.Result thenRes,
                                            RuleRow row,
                                            DslWriter dsl) {

    // 1) Extract from composed lines: exception/type codes on IF and THEN
    var ifExc  = extractCodesByKind(ifRes  == null ? null : ifRes.dslrLines,  "exception");
    var ifType = extractCodesByKind(ifRes  == null ? null : ifRes.dslrLines,  "type");
    var thExc  = extractCodesByKind(thenRes== null ? null : thenRes.dslrLines,"exception");
    var thType = extractCodesByKind(thenRes== null ? null : thenRes.dslrLines,"type");

    // 2) Apply merged CSV to THEN side if we truly merged
    String mergedCsv = row.mergedThenCodesCsv();
    boolean hasMerged = row.mergeGroupSize() > 1 && notBlank(mergedCsv);
    if (hasMerged) {
        var merged = splitCsv(mergedCsv);
        if (!thExc.isEmpty()) { thExc.clear(); thExc.addAll(merged); }
        else                  { thType.clear(); thType.addAll(merged); } // default to type if no explicit exc
    }

    // 3) If nothing to coalesce, fall back to original composed lines
    if (ifExc.isEmpty() && ifType.isEmpty() && thExc.isEmpty() && thType.isEmpty()) {
        if (ifRes   != null && ifRes.dslrLines   != null) for (String s : ifRes.dslrLines)   dsl.whenAnchor(s);
        if (thenRes != null && thenRes.dslrLines != null && !thenRes.dslrLines.isEmpty())
            dsl.thenAnchor(thenRes.dslrLines.get(0));
        return;
    }

    // 4) Emit: one WHEN header + dashes
    dsl.whenAnchor("Goods item with additional documents exists");

    // IF (positive) dashes
    if (!ifExc.isEmpty())  dsl.whenAnchor("- with exception code equals " + joinCsv(ifExc));
    if (!ifType.isEmpty()) dsl.whenAnchor("- with type code equals "      + joinCsv(ifType));

    // THEN (negated) dashes
    if (!thExc.isEmpty())  dsl.whenAnchor("- without exception code in " + joinCsv(thExc));
    if (!thType.isEmpty()) dsl.whenAnchor("- without type code in "      + joinCsv(thType));

    // 5) THEN header (keep your composed header)
    if (thenRes != null && thenRes.dslrLines != null && !thenRes.dslrLines.isEmpty()) {
        dsl.thenAnchor(thenRes.dslrLines.get(0));
    }
}


// Decide based on composed English
private boolean touchesAdditionalDocs(HyphenTwoPhaseComposer.Result ifRes,
                                      HyphenTwoPhaseComposer.Result thenRes) {
    return containsAdditionalDocs(ifRes) || containsAdditionalDocs(thenRes);
}
private boolean containsAdditionalDocs(HyphenTwoPhaseComposer.Result res) {
    if (res == null || res.dslrLines == null) return false;
    for (String s : res.dslrLines)
        if (s != null && s.toLowerCase().contains("additional document")) return true; // handles singular/plural
    return false;
}

// Pull codes for a given kind ("exception" or "type") from composed lines.
// Matches “… exception code … equals ABC” or “… type code … in A,B,C”
private java.util.LinkedHashSet<String> extractCodesByKind(java.util.List<String> lines, String kind) {
    java.util.LinkedHashSet<String> out = new java.util.LinkedHashSet<>();
    if (lines == null || lines.isEmpty()) return out;

    java.util.regex.Pattern p = java.util.regex.Pattern.compile(
        "(?i)\\b" + kind + "\\s*code\\b.*?\\b(?:equals|in)\\b\\s*([A-Za-z0-9_,\\-]+)"
    );

    for (String line : lines) {
        if (line == null) continue;
        java.util.regex.Matcher m = p.matcher(line);
        while (m.find()) {
            String g = m.group(1);
            if (g == null) continue;
            for (String tok : g.split("\\s*,\\s*")) {
                tok = tok.trim();
                if (!tok.isEmpty()) out.add(tok);
            }
        }
    }
    return out;
}

private java.util.LinkedHashSet<String> splitCsv(String csv) {
    java.util.LinkedHashSet<String> out = new java.util.LinkedHashSet<>();
    if (csv == null || csv.isBlank()) return out;
    for (String t : csv.split("\\s*,\\s*")) {
        t = t.trim();
        if (!t.isEmpty()) out.add(t);
    }
    return out;
}
private String joinCsv(java.util.LinkedHashSet<String> set) { return String.join(",", set); }
private boolean notBlank(String s){ return s != null && !s.isBlank(); }
