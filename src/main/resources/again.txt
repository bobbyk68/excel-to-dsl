public final class CompiledAtomic {
    private final String id;
    private final java.util.regex.Pattern regex;
    private final String pattern;   // your original pattern text
    private final String dsl;       // canonical English template
    private final uk.gov.hmrc.pattern.PatternIntrospector.Meta meta; // cached

    private CompiledAtomic(String id,
                           java.util.regex.Pattern regex,
                           String pattern,
                           String dsl,
                           uk.gov.hmrc.pattern.PatternIntrospector.Meta meta) {
        this.id = id;
        this.regex = regex;
        this.pattern = pattern;
        this.dsl = dsl;
        this.meta = meta;
    }

    public static CompiledAtomic compile(String id, String regexSource, String pattern, String dsl) {
        java.util.regex.Pattern rx = java.util.regex.Pattern.compile(regexSource);
        uk.gov.hmrc.pattern.PatternIntrospector.Meta meta =
            uk.gov.hmrc.pattern.PatternIntrospector.parse(dsl);

        // Safe defaults if parsing didn’t recognise the template/path
        if (meta == null) {
            meta = new uk.gov.hmrc.pattern.PatternIntrospector.Meta(
                    uk.gov.hmrc.pattern.PatternIntrospector.AnchorScope.GI, // assume GI as last resort
                    "GoodsItem",
                    uk.gov.hmrc.pattern.PatternIntrospector.FieldKey.UNKNOWN,
                    uk.gov.hmrc.pattern.PatternIntrospector.Operator.EQ,
                    uk.gov.hmrc.pattern.PatternIntrospector.Quantifier.EXISTS,
                    false,
                    uk.gov.hmrc.pattern.PatternIntrospector.PatternKind.UNKNOWN
            );
        }
        return new CompiledAtomic(id, rx, pattern, dsl, meta);
    }

    // getters used elsewhere
    public String id() { return id; }
    public java.util.regex.Pattern regex() { return regex; }
    public String pattern() { return pattern; }
    public String dsl() { return dsl; }
    public uk.gov.hmrc.pattern.PatternIntrospector.Meta meta() { return meta; }
}
