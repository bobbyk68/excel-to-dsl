// choose the declaration anchor path for the IF block
private String computeAnchorPathFromIf(List<ResultAdapters.ResolvedClause> list) {
    for (ResultAdapters.ResolvedClause c : list) {
        if (c.meta().anchorScope() == PatternIntrospector.AnchorScope.GI
                && c.meta().anchorPath() != null
                && c.meta().anchorPath().startsWith("GoodsItem")) {
            return c.meta().anchorPath();
        }
    }
    // Fallback from field labels (if we didn’t see an explicit anchor line)
    for (ResultAdapters.ResolvedClause c : list) {
        switch (c.meta().fieldKey()) {
            case AD_EXEMPTION_CODE:
            case AD_TYPE_CODE: return "GoodsItem.additionalDocuments";
            case SP_CODE:      return "GoodsItem.specialProcedures";
            case AI_CODE:      return "GoodsItem.additionalInformation";
            default: break;
        }
    }
    return null; // no GI context
}

// pick the first IF predicate line that actually has a field & value
private ResultAdapters.ResolvedClause firstPredicateForContext(List<ResultAdapters.ResolvedClause> list) {
    for (ResultAdapters.ResolvedClause c : list) {
        if (c.meta().fieldKey() != PatternIntrospector.FieldKey.UNKNOWN
                && c.value() != null && !c.value().isEmpty()) {
            return c;
        }
    }
    return null;
}


void emitAdditionalDocForCom(HyphenTwoPhaseComposer.Result ifRes,
                             HyphenTwoPhaseComposer.Result thenRes,
                             RuleRow row,
                             DslrWriter dslrWriter) {

    List<ResultAdapters.ResolvedClause> ifClauses   = ResultAdapters.resolve(ifRes);
    List<ResultAdapters.ResolvedClause> thenClauses = ResultAdapters.resolve(thenRes);
    if (ifClauses == null || ifClauses.isEmpty()) return;

    // 1) Anchor path from IF (works for pretty anchor lines)
    String anchorPath = computeAnchorPathFromIf(ifClauses);
    if (anchorPath == null) return; // anchored semantics need GI

    // 2) Predicate for the context (e.g., exemption == "UE")
    ResultAdapters.ResolvedClause ctxPred = firstPredicateForContext(ifClauses);
    if (ctxPred == null) return;

    dslrWriter.whenGoodsItemExists(anchorPath);
    dslrWriter.andWithEquals(ctxPred.meta().fieldKey(), ctxPred.value());

    // 3) THEN → anchored violation triggers (unchanged switch you already have)
    for (ResultAdapters.ResolvedClause p : thenClauses) {
        PatternIntrospector.PatternKind kind = p.meta().patternKind();
        Object field = p.meta().fieldKey();
        List<String> vals = p.values();

        switch (kind) {
            case MUST_NOT_ONE_OF:
            case ALL_MUST_NOT_ONE_OF:
                if (vals != null && !vals.isEmpty()) dslrWriter.andWithIn(field, vals);
                else if (p.value() != null)          dslrWriter.andWithEquals(field, p.value());
                break;

            case AT_LEAST_ONE_MUST_EQUAL:
                dslrWriter.andWithoutEquals(field, p.value());
                break;

            case ALL_MUST_EQUAL:
                dslrWriter.andWithNotEquals(field, p.value());
                break;

            case AT_LEAST_ONE_IN:
                if (vals != null && !vals.isEmpty()) dslrWriter.andWithoutIn(field, vals);
                else if (p.value() != null)          dslrWriter.andWithoutEquals(field, p.value());
                break;

            default:
                emitGenericAnchoredViolation(dslrWriter, p); // your existing fallback
        }
    }
}
