public static Meta parse(String english) {
    if (english == null) english = "";
    final String t = english.trim();
    final String lower = t.toLowerCase();

    // 1) Quantifier
    Quantifier q;
    if (lower.startsWith("all "))                           q = Quantifier.ALL;
    else if (lower.startsWith("none "))                     q = Quantifier.NONE;
    else if (lower.startsWith("exactly one "))              q = Quantifier.EXACTLY_ONE;
    else if (lower.startsWith("there is at least one ")
          || lower.startsWith("at least one "))            q = Quantifier.EXISTS;
    else                                                    q = Quantifier.EXISTS;

    // 2) Canonical path if present (Excel/raw form)
    java.util.regex.Matcher pm = java.util.regex.Pattern
        .compile("(GoodsItem(?:\\.[A-Za-z]+){1,3}\\.code|Declaration\\.[A-Za-z\\.]+)")
        .matcher(t);
    String canonicalPath = pm.find() ? pm.group(1) : "";

    // 3) FieldKey: prefer canonical, else English label
    FieldKey fk = fieldKeyOf(canonicalPath);
    if (fk == FieldKey.UNKNOWN) fk = fieldKeyFromLabel(lower);

    // 4) Anchor scope/path
    AnchorScope scope;
    String anchorPath;
    if (canonicalPath.startsWith("GoodsItem.")) {
        scope = AnchorScope.GI;
        anchorPath = computeAnchorPath(canonicalPath);
    } else {
        String byPhrase = anchorPathFromEnglish(lower);
        if (byPhrase != null) { scope = AnchorScope.GI; anchorPath = byPhrase; }
        else if (fk == FieldKey.AD_EXEMPTION_CODE || fk == FieldKey.AD_TYPE_CODE) {
            scope = AnchorScope.GI; anchorPath = "GoodsItem.additionalDocuments";
        } else if (fk == FieldKey.SP_CODE) {
            scope = AnchorScope.GI; anchorPath = "GoodsItem.specialProcedures";
        } else if (fk == FieldKey.AI_CODE) {
            scope = AnchorScope.GI; anchorPath = "GoodsItem.additionalInformation";
        } else {
            scope = AnchorScope.DECL; anchorPath = "Declaration";
        }
    }

    // 5) Operator + Negation (robust)
    boolean mentionsSet =
            lower.contains(" one of ") ||
            lower.contains(" one-of ") ||
            lower.matches(".*\\bin\\b.*"); // word-bounded " in "
    Operator op = mentionsSet ? Operator.IN : Operator.EQ;

    boolean hasNegationIn =
            lower.contains("must not") ||
            lower.contains(" not one of") ||
            lower.contains(" is not one of") ||
            lower.contains(" be not one of") ||
            lower.contains(" not in ") ||
            lower.startsWith("without ") ||
            lower.contains(" without ");

    boolean neg;
    if (op == Operator.IN) {
        neg = hasNegationIn; // NOT IN => neg=true
    } else {
        neg = lower.contains("must not equal")
           || lower.contains(" not equal")
           || lower.startsWith("without ");
    }

    // 6) Pattern kind
    PatternKind kind = classify(op, q, neg);

    return new Meta(scope, anchorPath, fk, op, q, neg, kind);
}

private static String computeAnchorPath(String canonicalPath) {
    if (canonicalPath != null && canonicalPath.startsWith("GoodsItem.")) {
        String[] p = canonicalPath.split("\\.");
        if (p.length >= 2) return p[0] + "." + p[1];
    }
    if (canonicalPath == null) return "GoodsItem";
    int dot = canonicalPath.indexOf('.');
    return dot > 0 ? canonicalPath.substring(0, dot) : canonicalPath;
}

private static FieldKey fieldKeyOf(String canonicalPath) {
    String s = canonicalPath == null ? "" : canonicalPath.toLowerCase();
    if (s.contains("additionaldocuments.exemption.code")) return FieldKey.AD_EXEMPTION_CODE;
    if (s.contains("additionaldocuments.type.code"))      return FieldKey.AD_TYPE_CODE;
    if (s.contains("specialprocedures.code"))             return FieldKey.SP_CODE;
    if (s.contains("additionalinformation.code"))         return FieldKey.AI_CODE;
    return FieldKey.UNKNOWN;
}

private static FieldKey fieldKeyFromLabel(String lower) {
    if (lower.contains("exemption code") || lower.contains("exception code")) return FieldKey.AD_EXEMPTION_CODE;
    if (lower.contains("type code") || lower.contains("document type"))       return FieldKey.AD_TYPE_CODE;
    if (lower.contains("special procedure code") || lower.contains("special-procedure code")) return FieldKey.SP_CODE;
    if (lower.contains("additional-information code") || lower.contains("additional information code")) return FieldKey.AI_CODE;
    return FieldKey.UNKNOWN;
}

private static String anchorPathFromEnglish(String lower) {
    if (lower.startsWith("goods item with additional document exists"))    return "GoodsItem.additionalDocuments";
    if (lower.startsWith("goods item with special procedure exists"))      return "GoodsItem.specialProcedures";
    if (lower.startsWith("goods item with additional information exists")) return "GoodsItem.additionalInformation";
    return null;
}

private static PatternKind classify(Operator op, Quantifier q, boolean neg) {
    if (op == Operator.IN && neg && q == Quantifier.ALL)     return PatternKind.ALL_MUST_NOT_ONE_OF;
    if (op == Operator.IN && neg)                            return PatternKind.MUST_NOT_ONE_OF;
    if (op == Operator.EQ && !neg && q == Quantifier.EXISTS) return PatternKind.AT_LEAST_ONE_MUST_EQUAL;
    if (op == Operator.EQ && !neg && q == Quantifier.ALL)    return PatternKind.ALL_MUST_EQUAL;
    if (op == Operator.IN && !neg && q == Quantifier.EXISTS) return PatternKind.AT_LEAST_ONE_IN;
    return PatternKind.UNKNOWN;
}

import java.util.*;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

public final class ResultAdapters {

    public static final class ResolvedClause {
        private final String raw;
        private final PatternIntrospector.Meta meta;
        private final List<String> values;

        public ResolvedClause(String raw, PatternIntrospector.Meta meta, List<String> values) {
            this.raw = raw; this.meta = meta; this.values = values;
        }
        public String raw() { return raw; }
        public PatternIntrospector.Meta meta() { return meta; }
        public List<String> values() { return values; }
        public String value() { return (values == null || values.isEmpty()) ? null : values.get(0); }
    }

    public static List<ResolvedClause> resolve(HyphenTwoPhaseComposer.Result res) {
        if (res == null || res.dslrLines == null) return Collections.emptyList();
        List<ResolvedClause> out = new ArrayList<ResolvedClause>(res.dslrLines.size());
        for (String line : res.dslrLines) {
            PatternIntrospector.Meta meta = PatternIntrospector.parse(line);
            List<String> vals = extractValues(line);
            out.add(new ResolvedClause(line, meta, vals));
        }
        return out;
    }

    static List<String> extractValues(String line) {
        if (line == null || line.trim().isEmpty()) return Collections.emptyList();

        // quoted values
        Matcher qm = Pattern.compile("\"([^\"]+)\"").matcher(line);
        List<String> quoted = new ArrayList<String>();
        while (qm.find()) quoted.add(qm.group(1).trim());
        if (!quoted.isEmpty()) return dedupeStable(quoted);

        String lower = line.toLowerCase();
        int idx;

        // equals tail
        if ((idx = lower.indexOf("equals ")) >= 0) {
            String tail = line.substring(idx + "equals ".length()).trim();
            if (tail.isEmpty()) return Collections.emptyList(); // avoid ""
            return dedupeStable(splitCodes(tail));
        }
        // one of tail
        if ((idx = lower.indexOf("one of ")) >= 0) {
            String tail = line.substring(idx + "one of ".length());
            return dedupeStable(splitCodes(tail));
        }
        // last ' in ' tail
        if ((idx = lower.lastIndexOf(" in ")) >= 0) {
            String tail = line.substring(idx + " in ".length());
            return dedupeStable(splitCodes(tail));
        }
        return Collections.emptyList();
    }

    private static List<String> splitCodes(String s) {
        List<String> out = new ArrayList<String>();
        for (String part : s.split("[,\\s]+")) {
            String t = part.trim();
            if (!t.isEmpty() && t.matches("[A-Za-z0-9._/-]+")) out.add(t);
        }
        return out;
    }

    private static List<String> dedupeStable(List<String> in) {
        if (in == null || in.isEmpty()) return Collections.emptyList();
        LinkedHashSet<String> set = new LinkedHashSet<String>(in);
        return new ArrayList<String>(set);
    }
}

package uk.gov.hmrc.dsl;

import java.util.LinkedHashSet;
import java.util.List;
import java.util.ArrayList;

public class DslrWriter {

    private final StringBuilder sb = new StringBuilder(256);
    private boolean whenStarted = false;

    public String content() { return sb.toString(); }
    public boolean isEmpty() { return sb.length() == 0; }
    public void reset() { sb.setLength(0); whenStarted = false; }

    public void whenGoodsItemExists(String anchorPath) {
        beginWhenIfNeeded();
        sb.append("  ").append(describeAnchor(anchorPath)).append('\n');
    }

    public void andWithEquals(Object fieldKey, String value) {
        line("with " + labelFor(fieldKey) + " equals " + quote(value));
    }
    public void andWithNotEquals(Object fieldKey, String value) {
        line("with " + labelFor(fieldKey) + " not equals " + quote(value));
    }
    public void andWithoutEquals(Object fieldKey, String value) {
        line("without " + labelFor(fieldKey) + " equals " + quote(value));
    }
    public void andWithIn(Object fieldKey, List<String> values) {
        line("with " + labelFor(fieldKey) + " in " + csv(values));
    }
    public void andWithNotIn(Object fieldKey, List<String> values) {
        line("with " + labelFor(fieldKey) + " not in " + csv(values));
    }
    public void andWithoutIn(Object fieldKey, List<String> values) {
        line("without " + labelFor(fieldKey) + " in " + csv(values));
    }
    public void andWithAllInOnly(Object fieldKey, List<String> values) {
        line("with all " + labelFor(fieldKey) + " in " + csv(values) + " only");
    }
    public void andWithAllEquals(Object fieldKey, String value) {
        line("with all " + labelFor(fieldKey) + " equals " + quote(value));
    }
    public void andWithPredicate(String englishPredicateLine) {
        line(englishPredicateLine);
    }

    private void beginWhenIfNeeded() {
        if (!whenStarted) { sb.append("when").append('\n'); whenStarted = true; }
    }
    private void line(String text) {
        beginWhenIfNeeded();
        sb.append("    ").append(text).append('\n');
    }

    private String describeAnchor(String anchorPath) {
        if (anchorPath == null || anchorPath.isBlank()) return "Goods item exists";
        String p = anchorPath.toLowerCase();
        if (p.contains("additionaldocuments"))   return "Goods item with additional document exists";
        if (p.contains("specialprocedures"))     return "Goods item with special procedure exists";
        if (p.contains("additionalinformation")) return "Goods item with additional information exists";
        return "Goods item exists";
    }

    private String labelFor(Object fieldKey) {
        if (fieldKey == null) return "field";
        String k = fieldKey.toString().trim().toUpperCase();
        if (k.endsWith("AD_EXEMPTION_CODE")) return "exemption code";
        if (k.endsWith("AD_TYPE_CODE"))      return "type code";
        if (k.endsWith("SP_CODE"))           return "special procedure code";
        if (k.endsWith("AI_CODE"))           return "additional-information code";
        if (k.endsWith("UNKNOWN"))           return "field";
        return k.contains("_") ? k.toLowerCase().replace('_',' ').trim() : fieldKey.toString();
    }

    private String quote(String value) {
        if (value == null) return "\"\"";
        String v = value.trim();
        boolean quoted = v.length() >= 2 && v.startsWith("\"") && v.endsWith("\"");
        return quoted ? v : "\"" + v.replace("\"", "\\\"") + "\"";
    }

    private String csv(List<String> values) {
        if (values == null || values.isEmpty()) return "\"\"";
        LinkedHashSet<String> set = new LinkedHashSet<String>(values.size());
        for (String s : values) {
            if (s != null && !s.isBlank()) set.add(s.trim());
        }
        StringBuilder b = new StringBuilder();
        boolean first = true;
        for (String s : set) {
            if (!first) b.append(',');
            b.append(quote(s));
            first = false;
        }
        return b.toString();
    }
}

void emitAdditionalDocForCom(HyphenTwoPhaseComposer.Result ifRes,
                             HyphenTwoPhaseComposer.Result thenRes,
                             RuleRow row,
                             DslrWriter dslrWriter) {

    List<ResolvedClause> ifClauses   = ResultAdapters.resolve(ifRes);
    List<ResolvedClause> thenClauses = ResultAdapters.resolve(thenRes);
    if (ifClauses == null || ifClauses.isEmpty()) return;

    String anchorPath = computeAnchorPathFromIf(ifClauses);
    if (anchorPath == null) return;
    ResolvedClause ctx = firstPredicateForContext(ifClauses);
    if (ctx == null) return;

    dslrWriter.whenGoodsItemExists(anchorPath);
    dslrWriter.andWithEquals(ctx.meta().fieldKey(), ctx.value());

    for (ResolvedClause p : thenClauses) {
        String rawLower = p.raw() == null ? "" : p.raw().toLowerCase();
        if (rawLower.startsWith("goods item with ") && rawLower.endsWith(" exists")) continue;

        boolean noValues = p.values() == null || p.values().isEmpty();
        boolean noValue  = p.value() == null || p.value().isEmpty();
        boolean unknown  = p.meta().fieldKey() == PatternIntrospector.FieldKey.UNKNOWN;
        if (unknown && noValues && noValue) continue;

        PatternIntrospector.PatternKind kind = p.meta().patternKind();
        Object field = p.meta().fieldKey();

        switch (kind) {
            case MUST_NOT_ONE_OF:
            case ALL_MUST_NOT_ONE_OF:
                if (!noValues) dslrWriter.andWithIn(field, p.values());
                else if (!noValue) dslrWriter.andWithEquals(field, p.value());
                break;

            case AT_LEAST_ONE_MUST_EQUAL:
                dslrWriter.andWithoutEquals(field, !noValue ? p.value() : (!noValues ? p.values().get(0) : ""));
                break;

            case AT_LEAST_ONE_IN:
                if (!noValues) dslrWriter.andWithoutIn(field, p.values());
                else if (!noValue) dslrWriter.andWithoutEquals(field, p.value());
                break;

            case ALL_MUST_EQUAL:
                dslrWriter.andWithNotEquals(field, !noValue ? p.value() : (!noValues ? p.values().get(0) : ""));
                break;

            default:
                emitGenericAnchoredViolation(dslrWriter, p); // keep your existing fallback
        }
    }
}

private String computeAnchorPathFromIf(List<ResolvedClause> list) {
    if (list == null) return null;
    for (ResolvedClause c : list) {
        String ap = c.meta().anchorPath();
        if (ap != null && ap.startsWith("GoodsItem")) return ap;
    }
    for (ResolvedClause c : list) {
        switch (c.meta().fieldKey()) {
            case AD_EXEMPTION_CODE:
            case AD_TYPE_CODE: return "GoodsItem.additionalDocuments";
            case SP_CODE:      return "GoodsItem.specialProcedures";
            case AI_CODE:      return "GoodsItem.additionalInformation";
            default: break;
        }
    }
    return null;
}

private ResolvedClause firstPredicateForContext(List<ResolvedClause> list) {
    if (list == null) return null;
    for (ResolvedClause c : list) {
        if (c.meta().fieldKey() != PatternIntrospector.FieldKey.UNKNOWN &&
            c.value() != null && !c.value().isEmpty()) {
            return c;
        }
    }
    return null;
}
