public void collectAll(List<RuleRow> rows, DslWriter dsl) {
    for (RuleRow row : rows) {

        // 🔹 SPECIAL CASE: additionalDocuments → coalesce after compose (no AtomicHit assumptions)
        if (touchesAdditionalDocs(row)) {
            emitAdditionalDocsQuick(row, dsl);
        } else {
            // 🔹 NORMAL PATH (leave as-is / your existing code)
            // If you normally use composer + matcher here, keep that code.
            // Minimal safe fallback:
            if (row.ifCondition() != null && !row.ifCondition().isBlank()) {
                dsl.whenAnchor(row.ifCondition());
            }
            if (row.thenCondition() != null && !row.thenCondition().isBlank()) {
                dsl.thenAnchor(row.thenCondition());
            }
        }

        dsl.endRule(row.id()); // keep your existing closer
    }
}

// Minimal post-compose fold for GoodsItem.additionalDocuments.* using raw IF/THEN strings only.
private void emitAdditionalDocsQuick(RuleRow row, DslWriter dsl) {
    String ifText   = safe(row.ifCondition());
    String thenText = safe(row.thenCondition());

    // Parse IF side (positive)
    java.util.LinkedHashSet<String> ifExc  = extractCodesFor(ifText,   "exception");
    java.util.LinkedHashSet<String> ifType = extractCodesFor(ifText,   "type");

    // Parse THEN side (negated)
    java.util.LinkedHashSet<String> thenExc  = extractCodesFor(thenText, "exception");
    java.util.LinkedHashSet<String> thenType = extractCodesFor(thenText, "type");

    // If we truly merged, prefer the merged CSV for the THEN value bucket that exists (or default to type)
    String mergedCsv = row.mergedThenCodesCsv();
    boolean hasMerged = row.mergeGroupSize() > 1 && notBlank(mergedCsv);
    if (hasMerged) {
        java.util.LinkedHashSet<String> merged = splitCsv(mergedCsv);
        if (!thenExc.isEmpty())       { thenExc.clear();  thenExc.addAll(merged); }
        else if (!thenType.isEmpty()) { thenType.clear(); thenType.addAll(merged); }
        else                          { thenType.addAll(merged); } // default bucket
    }

    // Nothing relevant? bail to avoid emitting empty blocks
    if (ifExc.isEmpty() && ifType.isEmpty() && thenExc.isEmpty() && thenType.isEmpty()) {
        // fall back to raw lines
        if (notBlank(ifText))   dsl.whenAnchor(ifText);
        if (notBlank(thenText)) dsl.thenAnchor(thenText);
        return;
    }

    // WHEN header (single main line)
    dsl.whenAnchor("Goods item with additional documents exists");

    // IF dashes (positive)
    if (!ifExc.isEmpty())  dsl.whenAnchor("- with exception code equals " + joinCsv(ifExc));
    if (!ifType.isEmpty()) dsl.whenAnchor("- with type code equals " + joinCsv(ifType));

    // THEN dashes (negated)
    if (!thenExc.isEmpty())  dsl.whenAnchor("- without exception code in " + joinCsv(thenExc));
    if (!thenType.isEmpty()) dsl.whenAnchor("- without type code in " + joinCsv(thenType));

    // THEN header — safest fallback is your original THEN text
    if (notBlank(thenText)) dsl.thenAnchor(thenText);
}


private static boolean touchesAdditionalDocs(RuleRow row) {
    String a = safe(row.ifCondition()).toLowerCase();
    String b = safe(row.thenCondition()).toLowerCase();
    return a.contains("additionaldocuments") || b.contains("additionaldocuments");
}
private static String safe(String s){ return s == null ? "" : s; }
private static boolean notBlank(String s){ return s != null && !s.isBlank(); }

private static java.util.LinkedHashSet<String> extractCodesFor(String text, String kind /* "exception" | "type" */) {
    java.util.LinkedHashSet<String> out = new java.util.LinkedHashSet<>();
    if (text == null || text.isBlank()) return out;
    // Look for "... exception code ... <LIST>" or "... type code ... <LIST>"
    java.util.regex.Pattern p = java.util.regex.Pattern.compile(
        "(?i)" + kind + "\\s*code\\s*(?:equals|in)?\\s*([A-Za-z0-9_,\\-]+)"
    );
    java.util.regex.Matcher m = p.matcher(text);
    while (m.find()) {
        String g = m.group(1);
        if (g != null) {
            for (String token : g.split("\\s*,\\s*")) {
                token = token.trim();
                if (!token.isEmpty()) out.add(token);
            }
        }
    }
    return out;
}

private static java.util.LinkedHashSet<String> splitCsv(String csv) {
    java.util.LinkedHashSet<String> out = new java.util.LinkedHashSet<>();
    if (csv == null || csv.isBlank()) return out;
    for (String t : csv.split("\\s*,\\s*")) {
        t = t.trim();
        if (!t.isEmpty()) out.add(t);
    }
    return out;
}
private static String joinCsv(java.util.LinkedHashSet<String> set) {
    return String.join(",", set);
}
