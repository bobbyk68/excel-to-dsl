class PatternIntrospector {

    enum AnchorScope { DECL, GI, SUBROW }
    enum Operator    { EQ, IN }
    enum Quantifier  { EXISTS, ALL, NONE, EXACTLY_ONE }
    enum FieldKey    { AD_EXEMPTION_CODE, AD_TYPE_CODE, SP_CODE, AI_CODE, UNKNOWN }
    enum PatternKind {
        MUST_NOT_ONE_OF, ALL_MUST_NOT_ONE_OF, AT_LEAST_ONE_MUST_EQUAL,
        ALL_MUST_EQUAL, AT_LEAST_ONE_IN, UNKNOWN
    }

    static final class Meta {
        final AnchorScope anchorScope;
        final String anchorPath;      // e.g. "GoodsItem.additionalDocuments"
        final FieldKey fieldKey;
        final Operator operator;
        final Quantifier quantifier;
        final boolean negated;
        final PatternKind patternKind;

        Meta(AnchorScope scope, String path, FieldKey fk, Operator op,
             Quantifier q, boolean neg, PatternKind kind) {
            this.anchorScope = scope; this.anchorPath = path; this.fieldKey = fk;
            this.operator = op; this.quantifier = q; this.negated = neg; this.patternKind = kind;
        }
    }

    // ──────────────────────────────────────────────────────────────
    // Parse a canonical English/DSL template (e.g. your ca.dsl()).
    // Robust to variants like "must is not one of".
    // ──────────────────────────────────────────────────────────────
    static Meta parse(String template) {
        String t = template.trim();

        // 1) Quantifier
        Quantifier q;
        if (t.toLowerCase().startsWith("all ")) q = Quantifier.ALL;
        else if (t.toLowerCase().startsWith("none ")) q = Quantifier.NONE;
        else if (t.toLowerCase().startsWith("exactly one ")) q = Quantifier.EXACTLY_ONE;
        else if (t.toLowerCase().startsWith("there is at least one ")
              || t.toLowerCase().startsWith("at least one ")) q = Quantifier.EXISTS;
        else q = Quantifier.EXISTS; // sensible default for single-line patterns

        // 2) Canonical path (we expect it to appear literally in your template)
        //    Examples we support:
        //    GoodsItem.additionalDocuments.exemption.code
        //    GoodsItem.additionalDocuments.type.code
        //    GoodsItem.specialProcedures.code
        //    GoodsItem.additionalInformation.code
        java.util.regex.Matcher pm = java.util.regex.Pattern
            .compile("(GoodsItem(?:\\.[A-Za-z]+){1,3}\\.code|Declaration\\.[A-Za-z\\.]+)")
            .matcher(t);
        String canonicalPath = pm.find() ? pm.group(1) : "";

        // 3) Anchor scope + anchor path
        AnchorScope scope = canonicalPath.startsWith("GoodsItem.") ? AnchorScope.GI : AnchorScope.DECL;
        String anchorPath = computeAnchorPath(canonicalPath); // collection-level for GI

        // 4) FieldKey mapping
        FieldKey fk = fieldKeyOf(canonicalPath);

        // 5) Operator + negation
        //    Accept "equals"/"equal", "one of"/"in", "must not"/"not one of", "without"
        String lower = t.toLowerCase();
        Operator op;
        boolean neg;
        if (lower.contains("one of") || lower.matches(".*\\bin\\b.*")) {
            op  = Operator.IN;
            neg = lower.contains("must not") || lower.contains("not one of") || lower.contains("without");
        } else {
            op  = Operator.EQ; // "equals"/"equal" or implied equality
            neg = lower.contains("must not equal") || lower.contains("not equal");
        }

        // 6) Classify to a pattern kind (helps your emitter switch deterministically)
        PatternKind kind = classify(op, q, neg, lower);

        return new Meta(scope, anchorPath, fk, op, q, neg, kind);
    }

    // ── helpers ───────────────────────────────────────────────────
    private static String computeAnchorPath(String canonicalPath) {
        if (canonicalPath.startsWith("GoodsItem.")) {
            String[] p = canonicalPath.split("\\.");
            // "GoodsItem.additionalDocuments.exemption.code" → "GoodsItem.additionalDocuments"
            // "GoodsItem.specialProcedures.code"            → "GoodsItem.specialProcedures"
            if (p.length >= 2) return p[0] + "." + p[1];
        }
        // Declaration-level: keep "Declaration" or first segment
        int dot = canonicalPath.indexOf('.');
        return dot > 0 ? canonicalPath.substring(0, dot) : canonicalPath;
    }

    private static FieldKey fieldKeyOf(String canonicalPath) {
        String s = canonicalPath.toLowerCase();
        if (s.contains("additionaldocuments.exemption.code")) return FieldKey.AD_EXEMPTION_CODE;
        if (s.contains("additionaldocuments.type.code"))      return FieldKey.AD_TYPE_CODE;
        if (s.contains("specialprocedures.code"))             return FieldKey.SP_CODE;
        if (s.contains("additionalinformation.code"))         return FieldKey.AI_CODE;
        return FieldKey.UNKNOWN;
    }

    private static PatternKind classify(Operator op, Quantifier q, boolean neg, String lowerTemplate) {
        if (op == Operator.IN && neg && q == Quantifier.ALL)    return PatternKind.ALL_MUST_NOT_ONE_OF;
        if (op == Operator.IN && neg)                           return PatternKind.MUST_NOT_ONE_OF;
        if (op == Operator.EQ && !neg && q == Quantifier.EXISTS) return PatternKind.AT_LEAST_ONE_MUST_EQUAL;
        if (op == Operator.EQ && !neg && q == Quantifier.ALL)    return PatternKind.ALL_MUST_EQUAL;
        if (op == Operator.IN && !neg && q == Quantifier.EXISTS) return PatternKind.AT_LEAST_ONE_IN;
        return PatternKind.UNKNOWN;
    }
}
