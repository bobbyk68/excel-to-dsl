// ─────────────────────────────────────────────────────────────────────────────
// CHANGED: Anchored policy→violation emitter (keeps same 4-param signature)
// Purpose:
//   • Bind SAME-GoodsItem anchor from IF.
//   • Convert THEN policies into violation triggers on the SAME GoodsItem.
//   • Stop deriving from raw text; use semantics carried by AtomicHit.
// Assumptions:
//   • HyphenComposer.Result provides a list of AtomicHit (getter shown as getHits()).
//   • AtomicHit exposes: anchorScope(), anchorPath(), fieldKey(), patternKind(),
//     values(), value(), quantifier(), operator(), isNegated().
//   • DslrWriter supports: whenGoodsItemExists(String path), andWithEquals(...),
//     andWithIn(...), andWithoutEquals(...), andWithNotEquals(...), andWithoutIn(...),
//     andWithNotIn(...), andWithAllInOnly(...), andWithAllEquals(...), andWithPredicate(String).
//   • Connectors (AND/OR/NOT) are resolved upstream by HyphenComposer (or handled elsewhere).
// ─────────────────────────────────────────────────────────────────────────────
void emitAdditionalDocForCom(HyphenComposer.Result ifRes,
                             HyphenComposer.Result thenRes,
                             RuleRow row,
                             DslrWriter dslrWriter) {

    // ── Guard: must have IF context to anchor on a GoodsItem
    java.util.List<AtomicHit> ifHits = hitsOf(ifRes);
    if (ifHits.isEmpty()) return;

    AtomicHit context = firstGoodsItemContext(ifHits);
    if (context == null) {
        // If you truly want to allow DECL-level only rules, handle here; for anchored semantics we skip.
        return;
    }

    // ── 1) Start WHEN … anchored to SAME GoodsItem as the IF clause
    dslrWriter.whenGoodsItemExists(context.anchorPath());
    dslrWriter.andWithEquals(context.fieldKey(), context.value());

    // ── 2) Convert THEN policies → anchored violation triggers
    for (AtomicHit p : hitsOf(thenRes)) {
        java.util.List<String> vals = dedupeStable(p.values());

        switch (p.patternKind()) {

            // “... type.code must not be one of S”
            case MUST_NOT_ONE_OF -> {
                // Violation (anchored): WITH type.code IN S
                if (!vals.isEmpty()) {
                    dslrWriter.andWithIn(p.fieldKey(), vals);
                } else if (p.value() != null) {
                    dslrWriter.andWithEquals(p.fieldKey(), p.value());
                }
            }

            // “ALL … code must not be one of S”  (counterexample)
            case ALL_MUST_NOT_ONE_OF -> {
                // Violation (anchored): WITH code IN S
                if (!vals.isEmpty()) {
                    dslrWriter.andWithIn(p.fieldKey(), vals);
                } else if (p.value() != null) {
                    dslrWriter.andWithEquals(p.fieldKey(), p.value());
                }
            }

            // “at least one … code must equal X”
            case AT_LEAST_ONE_MUST_EQUAL -> {
                // Violation (anchored): WITHOUT code == X  (absence of required value)
                dslrWriter.andWithoutEquals(p.fieldKey(), p.value());
            }

            // Optional explicit patterns (kept readable & explicit)
            case ALL_MUST_EQUAL -> {
                // Violation (anchored): WITH code != X  (counterexample)
                dslrWriter.andWithNotEquals(p.fieldKey(), p.value());
            }

            case AT_LEAST_ONE_IN -> {
                // Violation (anchored): WITHOUT any member in S
                if (!vals.isEmpty()) {
                    dslrWriter.andWithoutIn(p.fieldKey(), vals);
                } else if (p.value() != null) {
                    // single value form
                    dslrWriter.andWithoutEquals(p.fieldKey(), p.value());
                }
            }

            // Unknown/misc → generic inversion based on quantifier/operator/negation
            default -> emitGenericAnchoredViolation(dslrWriter, p, vals);
        }
    }
}

// ─────────────────────────────────────────────────────────────────────────────
// NEW (helper): choose first GI-scoped hit as the anchor (SAME GoodsItem).
// ─────────────────────────────────────────────────────────────────────────────
private AtomicHit firstGoodsItemContext(java.util.List<AtomicHit> hits) {
    for (AtomicHit h : hits) {
        if (h.anchorScope() == PatternIntrospector.AnchorScope.GI) return h;
        String path = h.anchorPath();
        if (path != null && path.startsWith("GoodsItem")) return h; // tolerant fallback
    }
    return null;
}

// ─────────────────────────────────────────────────────────────────────────────
// NEW (helper): generic inversion when patternKind is UNKNOWN or unhandled.
// Encodes the counterexample logic:
//   • ALL ∉ S  →  ∃ ∈ S        (withIn)
//   • ALL ∈ S  →  ∃ ∉ S        (withNotIn)
//   • ALL == X →  ∃ != X       (withNotEquals)
//   • ALL != X →  ∃ == X       (withEquals)
//   • ∃ == X   →  ¬∃ == X      (withoutEquals)
//   • ∃ ∈ S    →  ¬∃ ∈ S      (withoutIn)
// ─────────────────────────────────────────────────────────────────────────────
private void emitGenericAnchoredViolation(DslrWriter out, AtomicHit p, java.util.List<String> vals) {
    switch (p.quantifier()) {
        case ALL -> {
            if (p.isNegated() && p.operator() == PatternIntrospector.Operator.IN) {
                // ALL ∉ S → ∃ ∈ S
                if (!vals.isEmpty()) out.andWithIn(p.fieldKey(), vals);
                else out.andWithEquals(p.fieldKey(), p.value());
            } else if (!p.isNegated() && p.operator() == PatternIntrospector.Operator.IN) {
                // ALL ∈ S → ∃ ∉ S
                if (!vals.isEmpty()) out.andWithNotIn(p.fieldKey(), vals);
                else out.andWithNotEquals(p.fieldKey(), p.value());
            } else if (!p.isNegated() && p.operator() == PatternIntrospector.Operator.EQ) {
                // ALL == X → ∃ != X
                out.andWithNotEquals(p.fieldKey(), p.value());
            } else if (p.isNegated() && p.operator() == PatternIntrospector.Operator.EQ) {
                // ALL != X → ∃ == X
                out.andWithEquals(p.fieldKey(), p.value());
            } else {
                out.andWithPredicate(p.dsl()); // conservative fallback to positive predicate text
            }
        }
        case EXISTS, NONE, EXACTLY_ONE -> {
            // Treat EXISTS/NONE/EXACTLY_ONE as “some-kind-of existence requirement”:
            //   ∃ == X → NOT ∃ == X
            //   ∃ ∈ S  → NOT ∃ ∈ S
            if (!p.isNegated() && p.operator() == PatternIntrospector.Operator.EQ) {
                out.andWithoutEquals(p.fieldKey(), p.value());
            } else if (!p.isNegated() && p.operator() == PatternIntrospector.Operator.IN) {
                if (!vals.isEmpty()) out.andWithoutIn(p.fieldKey(), vals);
                else out.andWithoutEquals(p.fieldKey(), p.value());
            } else if (p.isNegated() && p.operator() == PatternIntrospector.Operator.IN) {
                // ∃ ∉ S → ALL ∈ S  (approximation; refine if you model counts)
                if (!vals.isEmpty()) out.andWithAllInOnly(p.fieldKey(), vals);
                else out.andWithPredicate(p.dsl());
            } else if (p.isNegated() && p.operator() == PatternIntrospector.Operator.EQ) {
                // ∃ != X → ALL == X
                out.andWithAllEquals(p.fieldKey(), p.value());
            } else {
                out.andWithPredicate(p.dsl());
            }
        }
        default -> out.andWithPredicate(p.dsl());
    }
}

// ─────────────────────────────────────────────────────────────────────────────
// NEW (helper): stable-dedupe values for IN lists (preserve author order).
// ─────────────────────────────────────────────────────────────────────────────
private java.util.List<String> dedupeStable(java.util.List<String> in) {
    if (in == null || in.isEmpty()) return java.util.Collections.emptyList();
    java.util.LinkedHashSet<String> set = new java.util.LinkedHashSet<>(in.size());
    for (String s : in) if (s != null && !s.isBlank()) set.add(s.trim());
    return new java.util.ArrayList<>(set);
}

// ─────────────────────────────────────────────────────────────────────────────
// NEW (helper): tolerant accessor for HyphenComposer.Result → List<AtomicHit>
// Adjust to your actual API if needed (e.g., res.hits() / res.getClauses()).
// ─────────────────────────────────────────────────────────────────────────────
private java.util.List<AtomicHit> hitsOf(HyphenComposer.Result res) {
    if (res == null) return java.util.Collections.emptyList();
    return res.getHits(); // ← rename if your API differs
}


void emitAdditionalDocForCom(HyphenComposer.Result ifRes,
                             HyphenComposer.Result thenRes,
                             RuleRow row,
                             DslrWriter out) {

    if (ifRes == null /* or ifRes.isEmpty() per your API */) return;

    // pick the first GoodsItem-scoped context hit
    AtomicHit context = selectGoodsItemContext(
        /* replace with your list getter */ ifRes.getHits()
    );
    if (context == null) {
        // fallback to first hit if you don't find a GI-scoped one
        List<AtomicHit> hits = ifRes.getHits();
        if (hits == null || hits.isEmpty()) return;
        context = hits.get(0);
    }

    // Start WHEN… anchored to the SAME GoodsItem as the context hit
    out.whenGoodsItemExists(
        /* anchor path for the GI scope */ context.getAnchorPath()
    );
    out.andWithEquals(
        /* canonical field key */ context.getFieldKey(),
        /* single value */       context.getValue()
    );

    // THEN policies → anchored violation triggers
    if (thenRes != null /* && !thenRes.isEmpty() per your API */) {
        for (AtomicHit p : thenRes.getHits()) {
            switch (p.getPatternKind()) {
                case MUST_NOT_ONE_OF -> {
                    // “... type.code must not be one of S” → WITH type.code IN S
                    out.andWithIn(p.getFieldKey(), p.getValues());
                }
                case ALL_MUST_NOT_ONE_OF -> {
                    // “ALL … code must not be one of S” → WITH code IN S (counterexample)
                    out.andWithIn(p.getFieldKey(), p.getValues());
                }
                case AT_LEAST_ONE_MUST_EQUAL -> {
                    // “… must equal X” → WITHOUT == X (absence = violation)
                    out.andWithoutEquals(p.getFieldKey(), p.getValue());
                }
                default -> {
                    // generic inversion (quantifier/operator/negation)
                    emitGenericAnchoredViolation(out, p);
                }
            }
        }
    }
}

// Choose a GI-scoped context hit
private AtomicHit selectGoodsItemContext(List<AtomicHit> hits) {
    if (hits == null) return null;
    for (AtomicHit h : hits) {
        // Replace with your own test for GoodsItem scope:
        // e.g., h.getAnchorScope()==GI or h.getAnchorPath().startsWith("GoodsItem")
        String path = h.getAnchorPath();
        if (path != null && path.startsWith("GoodsItem")) return h;
    }
    return null;
}

private void emitGenericAnchoredViolation(DslrWriter out, AtomicHit p) {
    switch (p.getQuantifier()) {
        case ALL -> {
            if (p.isNegated() && p.getOperator() == Operator.IN) {
                // ALL ∉ S → ∃ ∈ S
                out.andWithIn(p.getFieldKey(), p.getValues());
            } else if (!p.isNegated() && p.getOperator() == Operator.IN) {
                // ALL ∈ S → ∃ ∉ S
                out.andWithNotIn(p.getFieldKey(), p.getValues());
            } else if (!p.isNegated() && p.getOperator() == Operator.EQ) {
                // ALL == X → ∃ != X
                out.andWithNotEquals(p.getFieldKey(), p.getValue());
            } else if (p.isNegated() && p.getOperator() == Operator.EQ) {
                // ALL != X → ∃ == X
                out.andWithEquals(p.getFieldKey(), p.getValue());
            } else {
                out.andWithPredicate(p.toPositivePredicate()); // conservative fallback
            }
        }
        case AT_LEAST_ONE, EXISTS -> {
            if (!p.isNegated() && p.getOperator() == Operator.EQ) {
                // ∃ == X → NOT ∃ == X
                out.andWithoutEquals(p.getFieldKey(), p.getValue());
            } else if (!p.isNegated() && p.getOperator() == Operator.IN) {
                // ∃ ∈ S → NOT ∃ ∈ S
                out.andWithoutIn(p.getFieldKey(), p.getValues());
            } else if (p.isNegated() && p.getOperator() == Operator.IN) {
                // ∃ ∉ S → ALL ∈ S (approx; refine if you model counts)
                out.andWithAllInOnly(p.getFieldKey(), p.getValues());
            } else if (p.isNegated() && p.getOperator() == Operator.EQ) {
                // ∃ != X → ALL == X
                out.andWithAllEquals(p.getFieldKey(), p.getValue());
            } else {
                out.andWithPredicate(p.toPositivePredicate());
            }
        }
        default -> out.andWithPredicate(p.toPositivePredicate());
    }
}
