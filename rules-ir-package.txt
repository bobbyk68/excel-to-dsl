#!/usr/bin/env bash
set -euo pipefail

PKG_FILE="${1:-rules-ir-package.txt}"

if [[ ! -f "$PKG_FILE" ]]; then
  echo "Package file not found: $PKG_FILE" >&2
  exit 1
fi

current_file=""

while IFS= read -r line; do
  if [[ "$line" =~ ^====\ FILE:\ (.+)\ ==== ]]; then
    current_file="${BASH_REMATCH[1]}"
    echo "Writing $current_file"
    mkdir -p "$(dirname "$current_file")"
    : > "$current_file"
  else
    if [[ -n "$current_file" ]]; then
      printf '%s\n' "$line" >> "$current_file"
    fi
  fi
done < "$PKG_FILE"

echo "Done."
